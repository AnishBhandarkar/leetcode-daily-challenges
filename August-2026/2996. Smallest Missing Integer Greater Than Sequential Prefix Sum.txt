2996. Smallest Missing Integer Greater Than Sequential Prefix Sum


// Approach 1:


/**
 * @param {number[]} nums
 * @return {number}
 */
var missingInteger = function (nums) {
    let seqSum = nums[0];

    for (let i = 1; i < nums.length; i++) {
        if (nums[i] === nums[i - 1] + 1) {
            seqSum += nums[i];
        } else {
            break;
        }
    }

    const set = new Set(nums);

    while (true) {
        if (!set.has(seqSum)) {
            return seqSum;
        }
        seqSum++;
    }
};

/*
T(n) = O(n+k) // k is the gap to the missing integer.
S(n) = O(n)
*/

// Approach 2: Sorting.

/**
 * @param {number[]} nums
 * @return {number}
 */
var missingInteger = function (nums) {
    let seqSum = nums[0];

    for (let i = 1; i < nums.length; i++) {
        if (nums[i] === nums[i - 1] + 1) {
            seqSum += nums[i];
        } else {
            break;
        }
    }

    nums.sort((a, b) => a - b);

    for (let i = 0; i < nums.length; i++) {
        if (nums[i] < seqSum) {
            continue;
        }
        if (nums[i] === seqSum) {
            seqSum++;
        }
        if (nums[i] > seqSum) {
            break;
        }
    }

    return seqSum;
};

/*
T(n) = O(n)
S(n) = O(1)
*/
