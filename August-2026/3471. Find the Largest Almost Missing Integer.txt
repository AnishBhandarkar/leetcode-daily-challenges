3471. Find the Largest Almost Missing Integer


/**
 * @param {number[]} nums
 * @param {number} k
 * @return {number}
 */
var largestInteger = function (nums, k) {
    const map = new Map();
    const n = nums.length;

    for (let i = 0; i < n; i++) {
        map.set(nums[i], (map.get(nums[i]) || 0) + 1);
    }

    // if k === n
    if (k === n) {
        let maxVal = -1;
        for (let num of nums) {
            maxVal = Math.max(maxVal, num);
        }
        return maxVal;
    }

    if (k === 1) {
        let maxVal = -1;
        for (const [num, freq] of map.entries()) {
            if (freq === 1) {
                maxVal = Math.max(maxVal, num);
            }
        }
        return maxVal;
    }

    // if 1 < k < n
    let maxVal = -1;
    if (map.get(nums[0]) === 1) {
        maxVal = Math.max(maxVal, nums[0]);
    }

    if (map.get(nums[nums.length - 1]) === 1) {
        maxVal = Math.max(maxVal, nums[nums.length - 1]);
    }

    return maxVal;
};

/*
T(n) = O(n)
S(n) = O(n)
