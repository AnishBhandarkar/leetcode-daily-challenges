32. Longest Valid Parentheses


/**
 * @param {string} s
 * @return {number}
 */
var longestValidParentheses = function (s) {
    const stack = [];
    stack.push(-1);

    let maxLen = 0;

    for (let i = 0; i < s.length; i++) {
        if (s[i] === '(') {
            stack.push(i);
        } else {
            stack.pop();

            if (stack.length === 0) {
                stack.push(i);
            } else {
                let currLen = i - stack[stack.length - 1];
                maxLen = Math.max(maxLen, currLen);
            }
        }
    }

    return maxLen;
};

/*
T(n) = O(n)
S(n) = O(n)
*/
