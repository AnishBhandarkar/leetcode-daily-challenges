856. Score of Parentheses


/**
 * @param {string} s
 * @return {number}
 */
var scoreOfParentheses = function (s) {
    const stack = [];

    let score = 0;

    for (let ch of s) {
        if (ch === '(') {
            stack.push(0);
        } else {
            let inner = stack.pop();

            const score = inner === 0 ? 1 : 2 * inner;

            if (stack.length > 0) {
                stack[stack.length - 1] += score;
            } else {
                stack.push(score);
            }
        }
    }
    return stack[0];
};

/*
T(n) = O(n)
S(n) = O(n)
*/
