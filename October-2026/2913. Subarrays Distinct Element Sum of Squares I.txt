2913. Subarrays Distinct Element Sum of Squares I


/**
 * @param {number[]} nums
 * @return {number}
 */
var sumCounts = function (nums) {
    const len = nums.length;

    let result = 0;

    for (let i = 0; i < len; i++) {
        let distinctSet = new Set();
        for (let j = i; j < len; j++) {
            distinctSet.add(nums[j]);
            result += (distinctSet.size ** 2);
        }
    }

    return result;
};

/*
T(n) = O(n^2)
S(n) = O(n)
*/
