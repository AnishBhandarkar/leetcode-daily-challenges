2089. Find Target Indices After Sorting Array

/**
 * @param {number[]} nums
 * @param {number} target
 * @return {number[]}
 */
var targetIndices = function (nums, target) {
    nums.sort((a, b) => a - b);

    const targetIndices = [];

    for (let i = 0; i < nums.length; i++) {
        if (nums[i] === target) {
            targetIndices.push(i);
        }
    }

    return targetIndices;
};

/*
T(n) = O(n logn)
S(n) = O(1)
*/
