2778. Sum of Squares of Special Elements


/**
 * @param {number[]} nums
 * @return {number}
 */
var sumOfSquares = function (nums) {
    let specialSum = 0;
    const n = nums.length;

    for (let i = 1; i <= nums.length; i++) {
        if (n % i === 0) {
            specialSum += (nums[i - 1] ** 2);
        }
    }

    return specialSum;
};

/*
T(n) = O(n)
S(n) = O(1)
*/
