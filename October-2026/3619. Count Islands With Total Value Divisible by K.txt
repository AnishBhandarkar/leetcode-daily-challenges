3619. Count Islands With Total Value Divisible by K


/**
 * @param {number[][]} grid
 * @param {number} k
 * @return {number}
 */
var countIslands = function (grid, k) {
    const rows = grid.length;
    const cols = grid[0].length;

    let islandsCount = 0;

    const directions = [
        [-1, 0], [1, 0], [0, -1], [0, 1]
    ];

    const dfs = (r, c) => {
        if (r < 0 || c < 0 || r >= rows || c >= cols || grid[r][c] === 0) {
            return 0;
        }

        let sum = grid[r][c];

        grid[r][c] = 0; // visited

        for (const [dr, dc] of directions) {
            sum += dfs(r + dr, c + dc);
        }

        return sum;
    };

    for (let i = 0; i < rows; i++) {
        for (let j = 0; j < cols; j++) {
            if (grid[i][j] !== 0) {
                let totalSum = dfs(i, j);
                if (totalSum > 0 && totalSum % k === 0) {
                    islandsCount++;
                }
            }
        }
    }

    return islandsCount;
};

/*
T(n) = O(r * c)
S(n) = O(r * c);
*/
